----------------------------
SQL Query to extract data from FORS
----------------------------

SELECT
r.[id],
r.[ro_id],
r.[source_id],
r.[report_user_id],
r.[title],
r.[name],
r.[vol],
r.[year],
r.[category],
r.[sub_category],
c.[sub_category_fullname],
r.[output_type],
r.[published],
r.[accepted],
r.[input_date],
r.[report_year],
r.[owner],
r.[onbehalfof],
r.[unit] AS unit_ro,
r.[representative_pub],
r.[source_from],
a.[author_id],
a.[seq],
a.[last_name],
a.[first_name],
a.ust_staff,
a.pg_student,
a.[unit] AS unit_au,
a.[ust_email]
FROM ro_submitted_view r
INNER JOIN authors_submitted_view a ON r.ro_id=a.ro_id AND
r.source_id=a.source_id
INNER JOIN ro_sub_category_view c ON r.sub_category=c.sub_category_id
ORDER BY r.[ro_id],r.[id],a.[seq]
